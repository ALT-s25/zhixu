二、异常处理 ⭐⭐

2.1 基本语法

cpp
try {
    // 可能抛异常的代码
    throw std::runtime_error("出错了");
} catch (const std::runtime_error& e) {
    std::cerr << "运行时错误：" << e.what() << "\n";
} catch (const std::exception& e) {
    std::cerr << "其他异常：" << e.what() << "\n";
} catch (...) {
    std::cerr << "未知异常\n";
}

2.2 throw

cpp
// 抛出异常
throw std::runtime_error("something went wrong");
throw 42;                           // 抛 int
throw "error";                      // 抛 const char*

2.3 标准异常类 ⭐

text
exception
├── logic_error
│   ├── invalid_argument
│   ├── out_of_range
│   └── domain_error
└── runtime_error
    ├── overflow_error
    └── range_error

2.4 自定义异常

cpp
class MyException : public std::exception {
    std::string msg_;
public:
    MyException(const std::string& msg) : msg_(msg) {}
    const char* what() const noexcept override {
        return msg_.c_str();
    }
};

throw MyException("自定义错误");

2.5 noexcept ⭐

承诺函数不会抛出异常，编译器可以优化。

cpp
void func() noexcept {
    // 不抛异常
}

void risky() noexcept(false) {     // 显式声明可能抛
    throw std::runtime_error("");
}

// 条件 noexcept
template <typename T>
void swap(T& a, T& b) noexcept(std::is_nothrow_move_constructible_v<T>);

⚠ noexcept 函数抛异常 → std::terminate

2.6 异常最佳实践 ⭐

cpp
// 1. 析构函数绝对不能抛异常
~MyClass() noexcept { }

// 2. RAII 配合异常处理
{
    std::unique_ptr<int> p(new int(10));    // 异常时也自动释放
}

// 3. 构造函数抛异常 → 对象未构造完成

三、多线程基础（C++11）⭐⭐⭐

3.1 std::thread

cpp
#include <thread>

void task(int n) {
    std::cout << "Task " << n << " running\n";
}

int main() {
    // 1. 创建并启动线程
    std::thread t1(task, 1);
    std::thread t2(task, 2);

    // 2. 等待线程结束
    t1.join();           // 主线程阻塞等 t1
    t2.join();

    // 3. detach（分离，让线程后台运行）
    // std::thread t3(task, 3);
    // t3.detach();
}

⚠ join 或 detach 必须调用一次，否则析构会 std::terminate。

3.2 Lambda 创建线程

cpp
std::thread t([]() {
    std::cout << "Hello from thread\n";
});
t.join();

3.3 std::async（C++11）⭐⭐⭐

异步任务，获取执行结果。

cpp
#include <future>

std::future<int> fut = std::async([]() {
    std::this_thread::sleep_for(std::chrono::seconds(1));
    return 42;
});

std::cout << "等待结果...\n";
int result = fut.get();     // 阻塞等待
std::cout << "结果：" << result;

启动策略：

cpp
std::launch::async    // 立即在新线程中执行
std::launch::deferred // 延迟到 get() 时才执行（同一线程）
std::launch::async | std::launch::deferred   // 默认（让系统选）

3.4 std::this_thread

cpp
std::this_thread::get_id();                       // 当前线程 id
std::this_thread::sleep_for(std::chrono::ms(100));// 休眠
std::this_thread::sleep_until(time_point);        // 休眠到指定时间
std::this_thread::yield();                        // 让出 CPU

3.5 std::future / std::promise

cpp
std::promise<int> prom;
std::future<int> fut = prom.get_future();

std::thread t([&prom]() {
    prom.set_value(42);   // 设置值
});

std::cout << fut.get();   // 阻塞获取
t.join();

四、互斥锁 ⭐⭐⭐

4.1 std::mutex

cpp
#include <mutex>

std::mutex mtx;
int counter = 0;

void increment() {
    mtx.lock();
    counter++;
    mtx.unlock();
}

// 推荐：使用 RAII 包装
{
    std::lock_guard<std::mutex> lock(mtx);
    counter++;
}   // 自动 unlock

4.2 std::lock_guard（RAII）⭐⭐

cpp
{
    std::lock_guard<std::mutex> lock(mtx);    // 加锁
    counter++;
}   // 析构自动解锁

💡 永远不要忘记解锁，用 lock_guard 避免忘记。

4.3 std::unique_lock ⭐⭐

比 lock_guard 更灵活，可延迟加锁、提前解锁。

cpp
std::unique_lock<std::mutex> lock(mtx);   // 加锁
// ... 中间可 lock.unlock() / lock.lock()

4.4 std::scoped_lock（C++17）⭐⭐

同时锁多个互斥量，避免死锁。

cpp
std::mutex mtx1, mtx2;

{
    std::scoped_lock lock(mtx1, mtx2);    // 同时锁多个
    // ...
}   // 自动按相反顺序解锁

4.5 try_lock / try_lock_for

cpp
std::timed_mutex tmtx;

if (tmtx.try_lock_for(std::chrono::ms(100))) {
    // 拿到锁了
    tmtx.unlock();
} else {
    // 超时了
}

五、条件变量 ⭐⭐

cpp
#include <condition_variable>

std::mutex mtx;
std::condition_variable cv;
bool ready = false;

void wait_for_ready() {
    std::unique_lock<std::mutex> lock(mtx);
    cv.wait(lock, []() { return ready; });   // 等待 ready 变 true
    std::cout << "已就绪\n";
}

void set_ready() {
    {
        std::lock_guard<std::mutex> lock(mtx);
        ready = true;
    }
    cv.notify_one();   // 通��一个等待者
}

std::thread t1(wait_for_ready);
std::this_thread::sleep_for(std::chrono::seconds(1));
std::thread t2(set_ready);

⚠ cv.wait 会自动释放锁并阻塞，被唤醒后重新加锁。

六、原子操作 ⭐

cpp
#include <atomic>

std::atomic<int> counter(0);

void increment() {
    counter.fetch_add(1);    // 原子加
}

// 整型、指针等都可 atomic
std::atomic<bool> flag(false);
flag.store(true);
bool b = flag.load();

💡 原子操作比互斥锁快，但只适合简单的读写。

// 多线程中只执行一次

八、线程池（C++ 本身没有，但可自己实现）

cpp
class ThreadPool {
    std::vector<std::thread> workers_;
    std::queue<std::function<void()>> tasks_;
    std::mutex mtx_;
    std::condition_variable cv_;
    bool stop_ = false;

public:
    ThreadPool(size_t n) {
        for (size_t i = 0; i < n; i++) {
            workers_.emplace_back([this] {
                while (true) {
                    std::function<void()> task;
                    {
                        std::unique_lock<std::mutex> lock(mtx_);
                        cv_.wait(lock, [this] {
                            return stop_ || !tasks_.empty();
                        });
                        if (stop_ && tasks_.empty()) return;
                        task = std::move(tasks_.front());
                        tasks_.pop();
                    }
                    task();
                }
            });
        }
    }

    template <typename F, typename... Args>
    auto enqueue(F&& f, Args&&... args) {
        using R = std::invoke_result_t<F, Args...>;
        auto task = std::make_shared<std::packaged_task<R()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        std::future<R> res = task->get_future();
        {
            std::lock_guard<std::mutex> lock(mtx_);
            tasks_.emplace([task]() { (*task)(); });
        }
        cv_.notify_one();
        return res;
    }

    ~ThreadPool() {
        {
            std::lock_guard<std::mutex> lock(mtx_);
            stop_ = true;
        }
        cv_.notify_all();
        for (auto& w : workers_) w.join();
    }
};

💡 现代项目推荐使用 Asio、QtThreadPool、Boost.Asio 等现成库。

九、面试常问 ⭐⭐⭐

💡 std::async vs std::thread？ | std::async | std::thread | |------------|-------------| | 自动管理线程 | 手动 join/detach | | 返回 future | 无返回值（用 promise） | | 可指定启动策略 | - | | 推荐 | - |

💡 std::lock_guard vs std::unique_lock？ | lock_guard | unique_lock | |------------|-------------| | 简单 | 灵活 | | 构造加锁、析构解锁 | 可延迟加锁、提前解锁 | | 首选 | 需要条件变量等场景 |

💡 如何避免死锁？

  • 固定加锁顺序
  • 用 std::lock() 或 std::scoped_lock
  • 减少嵌套锁
  • 用 try_lock

💡 为什么析构函数要 noexcept？ 栈展开时若析构函数抛异常，会导致 std::terminate 程序崩溃。

💡 原子操作 vs 互斥锁？

  • 原子操作：仅适合单变量读写（无锁，CAS）
  • 互斥锁：保护临界区（有锁，开销大）

💡 std::future::get 只能调一次？ 是的，get() 调用后 future 状态变为 ready。多次调用未定义。
