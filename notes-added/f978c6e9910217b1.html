MySQL DML（数据增删改）完整详解

按章节学习你的知识，慢慢成林

DML = Data Manipulation Language，即数据操作语言 专门用于对表中的数据进行增（INSERT）、改（UPDATE）、删（DELETE）操作 注意：DML 操作的是数据行，不是表结构（表结构用 DDL）

一、知识框架总览

text
DML
├── INSERT  插入数据
│   ├── 单行插入
│   ├── 多行插入
│   ├── 子查询插入
│   ├── INSERT IGNORE（忽略错误）
│   └── ON DUPLICATE KEY UPDATE（存在则更新）
│
├── UPDATE  修改数据
│   ├── 单表更新
│   ├── WHERE 条件过滤
│   └── UPDATE JOIN 多表联合更新
│
└── DELETE  删除数据
    ├── 单表删除
    ├── DELETE JOIN 多表联合删除
    └── TRUNCATE vs DELETE 对比

二、环境准备：示例表结构

以下所有示例都基于这三张表，请先创建好

sql
-- 创建用户表
CREATE TABLE users (
    id       INT PRIMARY KEY AUTO_INCREMENT,  -- 主键，自增
    name     VARCHAR(50) NOT NULL,            -- 姓名，不能为空
    email    VARCHAR(100) UNIQUE,             -- 邮箱，唯一约束
    age      INT DEFAULT 18,                  -- 年龄，默认18
    created_at DATETIME DEFAULT NOW()         -- 创建时间，默认当前时间
);

-- 创建商品表
CREATE TABLE products (
    id       INT PRIMARY KEY AUTO_INCREMENT,  -- 主键，自增
    name     VARCHAR(100) NOT NULL,           -- 商品名称
    price    DECIMAL(10, 2) NOT NULL,         -- 价格，2位小数
    stock    INT DEFAULT 0,                   -- 库存，默认0
    category VARCHAR(50)                      -- 分类
);

-- 创建订单表
CREATE TABLE orders (
    id         INT PRIMARY KEY AUTO_INCREMENT, -- 订单主键
    user_id    INT NOT NULL,                   -- 用户ID（外键）
    product_id INT NOT NULL,                   -- 商品ID（外键）
    quantity   INT NOT NULL DEFAULT 1,         -- 购买数量
    total      DECIMAL(10, 2),                 -- 订单总价
    status     VARCHAR(20) DEFAULT 'pending',  -- 订单状态
    created_at DATETIME DEFAULT NOW(),         -- 下单时间
    FOREIGN KEY (user_id) REFERENCES users(id),       -- 外键约束
    FOREIGN KEY (product_id) REFERENCES products(id)  -- 外键约束
);

三、INSERT：插入数据

3.1 单行插入

语法格式

sql
-- 方式一：指定列名（推荐！列名和值对应）
INSERT INTO 表名 (列1, 列2, ...) VALUES (值1, 值2, ...);

-- 方式二：不指定列名（必须给所有列赋值，包括AUTO_INCREMENT列写0或NULL）
INSERT INTO 表名 VALUES (值1, 值2, ...);

完整示例

sql
-- 示例1：指定列名插入用户（推荐写法）
INSERT INTO users (name, email, age)
VALUES ('张三', 'zhangsan@example.com', 25);
-- id 和 created_at 会自动填充（AUTO_INCREMENT 和 DEFAULT）

-- 示例2：不指定列名（必须按照表结构顺序填写所有列）
-- id 写 NULL 让它自增，created_at 写 NOW() 获取当前时间
INSERT INTO users
VALUES (NULL, '李四', 'lisi@example.com', 30, NOW());

-- 示例3：插入商品
INSERT INTO products (name, price, stock, category)
VALUES ('苹果手机', 6999.00, 100, '手机');

-- 示例4：只插入必填列，其他用默认值
INSERT INTO products (name, price)
VALUES ('华为手机', 5999.00);
-- stock 会用默认值 0，category 为 NULL

3.2 多行插入

一条 SQL 插入多行，比多条 SQL 执行效率更高

语法格式

sql
INSERT INTO 表名 (列1, 列2, ...) VALUES
    (值1, 值2, ...),  -- 第1行
    (值1, 值2, ...),  -- 第2行
    (值1, 值2, ...);  -- 第3行（末尾是分号）

完整示例

sql
-- 一次插入多个用户
INSERT INTO users (name, email, age) VALUES
    ('王五',   'wangwu@example.com',   28),    -- 第1行
    ('赵六',   'zhaoliu@example.com',  22),    -- 第2行
    ('孙七',   'sunqi@example.com',    35),    -- 第3行
    ('周八',   'zhouba@example.com',   19);    -- 第4行

-- 一次插入多个商品
INSERT INTO products (name, price, stock, category) VALUES
    ('小米手机',   2999.00, 200, '手机'),
    ('iPad',       4999.00,  50, '平板'),
    ('MacBook',   12999.00,  30, '电脑'),
    ('AirPods',    1299.00, 500, '耳机');

3.3 子查询插入（INSERT INTO ... SELECT）

把一张表的查询结果，插入到另一张表中

语法格式

sql
INSERT INTO 目标表 (列1, 列2, ...)
SELECT 列1, 列2, ...
FROM 来源表
WHERE 条件;

完整示例

sql
-- 场景：把用户表中年龄大于30的用户复制到 vip_users 表
-- 假设 vip_users 表已经存在，结构与 users 相同

-- 先创建 vip_users 表（结构参考 users）
CREATE TABLE vip_users LIKE users;  -- 复制表结构

-- 把 users 中 age > 30 的用户插入到 vip_users
INSERT INTO vip_users (name, email, age)
SELECT name, email, age    -- 从 users 表查询这三列
FROM users                 -- 来源表
WHERE age > 30;            -- 只取年龄大于30的

-- 结果：users 表中符合条件的记录被复制到了 vip_users

3.4 INSERT IGNORE：忽略重复错误

如果插入的数据违反了唯一约束（UNIQUE），正常情况下会报错并终止 加了 IGNORE 后，会静默跳过冲突的行，继续插入其他行

语法格式

sql
INSERT IGNORE INTO 表名 (列1, 列2) VALUES (值1, 值2);

完整示例

sql
-- 场景：users 表中 email 有唯一约束
-- 假设 zhangsan@example.com 已经存在

-- 不加 IGNORE：会报错 Duplicate entry，整个插入失败
-- INSERT INTO users (name, email) VALUES ('张三2', 'zhangsan@example.com');
-- ERROR 1062 (23000): Duplicate entry 'zhangsan@example.com' for key 'email'

-- 加了 IGNORE：冲突行被跳过，不报错，其他行正常插入
INSERT IGNORE INTO users (name, email, age) VALUES
    ('张三2', 'zhangsan@example.com', 26),  -- 邮箱重复，被跳过
    ('新用户', 'newuser@example.com',  20); -- 正常插入

-- 执行结果：只有"新用户"被插入，"张三2"被忽略（Query OK, 1 row affected）

3.5 INSERT INTO ... ON DUPLICATE KEY UPDATE

核心逻辑：有则更新，无则插入（Upsert 操作） 当插入的数据触发了唯一键或主键冲突时，不报错，改为执行 UPDATE

语法格式

sql
INSERT INTO 表名 (列1, 列2, ...)
VALUES (值1, 值2, ...)
ON DUPLICATE KEY UPDATE
    列1 = 新值1,
    列2 = 新值2;

完整示例

sql
-- 场景：用户登录系统，记录每个用户的登录次数和最后登录时间
-- 创建登录记录表
CREATE TABLE user_login_log (
    user_id    INT PRIMARY KEY,          -- 用户ID（主键，唯一）
    login_count INT DEFAULT 0,           -- 登录次数
    last_login  DATETIME                 -- 最后登录时间
);

-- 第一次插入用户1的登录记录（表中没有，执行 INSERT）
INSERT INTO user_login_log (user_id, login_count, last_login)
VALUES (1, 1, NOW())
ON DUPLICATE KEY UPDATE
    login_count = login_count + 1,  -- 登录次数 +1
    last_login  = NOW();            -- 更新最后登录时间

-- 第二次同样执行这条SQL（user_id=1 已存在，执行 UPDATE）
-- 结果：login_count 变成 2，last_login 更新为当前时间

-- 使用 VALUES() 函数引用要插入的值（MySQL 8.0.20+ 推荐用别名方式）
INSERT INTO products (id, name, price, stock)
VALUES (1, '苹果手机', 6999.00, 150)   -- 假设 id=1 已存在
ON DUPLICATE KEY UPDATE
    stock = VALUES(stock),    -- stock 更新为插入语句中指定的 150
    price = VALUES(price);    -- price 也更新

四、UPDATE：更新数据

4.1 单表更新

语法格式

sql
UPDATE 表名
SET 列1 = 值1, 列2 = 值2, ...
WHERE 条件;         -- ⚠️ 强烈建议加 WHERE！不加会更新所有行！

完整示例

sql
-- 示例1：更新单个字段（把 id=1 的用户年龄改为 26）
UPDATE users
SET age = 26          -- 修改 age 字段
WHERE id = 1;         -- 只更新 id=1 的行

-- 示例2：同时更新多个字段
UPDATE users
SET age   = 28,                        -- 修改年龄
    email = 'newemail@example.com'     -- 修改邮箱
WHERE id = 1;                          -- 条件：id=1

-- 示例3：用当前值计算新值（商品降价10%）
UPDATE products
SET price = price * 0.9    -- 新价格 = 旧价格 × 0.9（打9折）
WHERE category = '手机';   -- 只更新手机类商品

-- 示例4：更新库存（进货了，库存增加）
UPDATE products
SET stock = stock + 50    -- 库存增加50
WHERE id = 1;             -- 只更新 id=1 的商品

-- 示例5：批量更新所有手机的分类名称
UPDATE products
SET category = 'smartphone'    -- 把"手机"改为"smartphone"
WHERE category = '手机';

-- ⚠️ 危险！不加 WHERE 会更新所有行
-- UPDATE products SET price = 0;  -- 所有商品价格变成0！千万别这样！

4.2 UPDATE JOIN：多表联合更新

根据一张表的数据，更新另一张表（需要两表有关联关系）

语法格式

sql
UPDATE 表1
JOIN 表2 ON 表1.关联列 = 表2.关联列
SET 表1.列 = 新值
WHERE 条件;

完整示例

sql
-- 场景1：把 VIP 用户（vip_users 表中的用户）的年龄全部+1
UPDATE users u                          -- u 是 users 的别名
JOIN vip_users v ON u.id = v.id        -- 关联条件：两表 id 相同
SET u.age = u.age + 1;                 -- 更新 users 表的 age

-- 场景2：根据订单信息更新商品库存（下单后减库存）
-- orders 表中 status='completed' 的订单，减去对应商品的 quantity
UPDATE products p                                 -- p 是 products 别名
JOIN orders o ON p.id = o.product_id             -- 关联商品ID
SET p.stock = p.stock - o.quantity               -- 库存减去订单数量
WHERE o.status = 'completed';                    -- 只处理已完成的订单

-- 场景3：LEFT JOIN（无论右表是否有匹配都更新左表）
-- 把没有下过订单的用户的 age 重置为 0（示例，实际不这样用）
UPDATE users u
LEFT JOIN orders o ON u.id = o.user_id    -- 左连接
SET u.age = 0
WHERE o.id IS NULL;    -- o.id 为 NULL 表示该用户没有订单

五、DELETE：删除数据

5.1 单表删除

语法格式

sql
DELETE FROM 表名
WHERE 条件;    -- ⚠️ 强烈建议加 WHERE！不加会删除所有行！

完整示例

sql
-- 示例1：删除指定 id 的用户
DELETE FROM users
WHERE id = 1;    -- 删除 id=1 的用户

-- 示例2：删除满足条件的多行（删除年龄小于18的用户）
DELETE FROM users
WHERE age < 18;

-- 示例3：删除指定范围（删除 id 在 10到20 之间的记录）
DELETE FROM users
WHERE id BETWEEN 10 AND 20;    -- 删除 id 10~20 的记录

-- 示例4：删除库存为0的商品
DELETE FROM products
WHERE stock = 0;    -- 删除没有库存的商品

-- 示例5：配合 LIMIT 限制删除行数（安全删除，防止误删太多）
DELETE FROM users
WHERE age > 60
LIMIT 10;    -- 最多删除10行，避免误删大量数据

-- ⚠️ 危险！删除全部数据
-- DELETE FROM users;  -- 会删掉所有用户，慎用！

5.2 DELETE JOIN：多表联合删除

同时从多张表删除数据（表间有关联关系）

语法格式

sql
-- 只删除表1
DELETE 表1 FROM 表1
JOIN 表2 ON 表1.关联列 = 表2.关联列
WHERE 条件;

-- 同时删除表1和表2中的数据
DELETE 表1, 表2 FROM 表1
JOIN 表2 ON 表1.关联列 = 表2.关联列
WHERE 条件;

完整示例

sql
-- 场景1：删除指定用户及其所有订单（同时删两张表）
DELETE u, o          -- 要删除的表：u（users）和 o（orders）
FROM users u                              -- 主表（带别名）
JOIN orders o ON u.id = o.user_id        -- 关联订单表
WHERE u.id = 5;                          -- 删除 id=5 的用户及其订单

-- 场景2：只删除孤立的订单（商品已经被删除的订单）
-- 商品不存在了，但订单还在，清理这些脏数据
DELETE o               -- 只删除 orders 表数据
FROM orders o
LEFT JOIN products p ON o.product_id = p.id    -- 左连接商品表
WHERE p.id IS NULL;                            -- 商品不存在（左连接结果为NULL）

-- 场景3：删除重复数据（保留最小 id，删除其他重复 email）
-- 注意：这里用自连接，MySQL 中 DELETE 不能直接对被删表做子查询
-- 需要用临时表方式处理（进阶用法）
DELETE u1 FROM users u1
JOIN users u2                             -- 自连接
ON u1.email = u2.email                   -- 相同邮箱
AND u1.id > u2.id;                       -- 保留 id 小的，删除 id 大的

5.3 TRUNCATE vs DELETE

核心区别对比表

对比维度 DELETE TRUNCATE
语法 DELETE FROM 表名 WHERE 条件 TRUNCATE TABLE 表名
能加 WHERE ✅ 可以加条件，精准删除 ❌ 不能加条件，全表清空
事务 ✅ 支持事务，可回滚（ROLLBACK） ❌ 不支持事务，无法回滚
触发器 ✅ 会触发 DELETE 触发器 ❌ 不触发触发器
自增重置 ❌ 不重置 AUTO_INCREMENT ✅ 重置 AUTO_INCREMENT 为 1
执行速度 较慢（逐行删除，记录日志） 极快（直接重建表）
返回行数 返回实际删除的行数 返回 0（不统计行数）
外键约束 受外键约束限制 有外键时不能执行
DDL/DML DML（数据操作语言） DDL（数据定义语言）

完整示例

sql
-- DELETE 示例
DELETE FROM users;           -- 删除所有行，AUTO_INCREMENT 不重置
-- 假设之前 id 到了 100，下次插入的 id 还是 101

-- TRUNCATE 示例
TRUNCATE TABLE users;        -- 清空所有行，AUTO_INCREMENT 重置为 1
-- 下次插入的 id 从 1 开始

-- 验证区别（可自己测试）
INSERT INTO users (name, email) VALUES ('测试', 'test@test.com');
-- DELETE 后插入，id 可能是 101
-- TRUNCATE 后插入，id 一定是 1

六、常见报错及解决方法

6.1 Data truncated for column

text
ERROR 1265 (01000): Data truncated for column 'age' at row 1

原因：插入的数据超出了列的范围或类型不匹配

sql
-- 错误示例（age 是 INT，但插入了字符串）
INSERT INTO users (name, age) VALUES ('张三', 'abc');  -- age 不能是字符串

-- 错误示例（age 是 INT，但插入了浮点数——会被截断）
INSERT INTO users (name, age) VALUES ('张三', 25.9);   -- 25.9 被截断为 25

-- 解决：确保数据类型正确
INSERT INTO users (name, age) VALUES ('张三', 25);     -- 正确

6.2 Cannot add or update a child row: a foreign key constraint fails

text
ERROR 1452 (23000): Cannot add or update a child row: a foreign key constraint fails
(`mydb`.`orders`, CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`))

原因：插入订单时，user_id 在 users 表中不存在（外键约束）

sql
-- 错误示例（users 表中没有 id=999 的用户）
INSERT INTO orders (user_id, product_id, quantity, total)
VALUES (999, 1, 2, 100.00);    -- user_id=999 不存在，报错

-- 解决方法1：先确认用户存在
SELECT id FROM users WHERE id = 999;    -- 先查一下

-- 解决方法2：先插入用户，再插入订单
INSERT INTO users (name, email) VALUES ('新用户', 'new@test.com');
SET @new_user_id = LAST_INSERT_ID();    -- 获取刚插入的用户ID
INSERT INTO orders (user_id, product_id, quantity, total)
VALUES (@new_user_id, 1, 2, 100.00);   -- 用真实存在的 user_id

-- 解决方法3（临时方案，不推荐用于生产）：关闭外键检查
SET FOREIGN_KEY_CHECKS = 0;    -- 关闭外键检查
INSERT INTO orders ...;         -- 执行插入
SET FOREIGN_KEY_CHECKS = 1;    -- 重新开启外键检查

6.3 Duplicate entry for key

text
ERROR 1062 (23000): Duplicate entry 'zhangsan@example.com' for key 'email'

原因：违反了唯一约束（UNIQUE KEY）

sql
-- 解决方法1：使用 INSERT IGNORE（静默跳过）
INSERT IGNORE INTO users (name, email) VALUES ('张三2', 'zhangsan@example.com');

-- 解决方法2：使用 ON DUPLICATE KEY UPDATE（冲突时更新）
INSERT INTO users (name, email, age)
VALUES ('张三2', 'zhangsan@example.com', 26)
ON DUPLICATE KEY UPDATE age = 26;    -- 邮箱冲突时，只更新年龄

-- 解决方法3：先查后插（代码层判断）
SELECT id FROM users WHERE email = 'zhangsan@example.com';
-- 如果有结果，说明存在，不再插入

6.4 You are using safe update mode

text
ERROR 1175 (HY000): You are using safe update mode and you tried to update a table
without a WHERE that uses a KEY column

原因：MySQL 安全更新模式下，不允许没有主键条件的 UPDATE/DELETE

sql
-- 临时关闭安全模式（当前会话有效）
SET SQL_SAFE_UPDATES = 0;

-- 执行你的更新
UPDATE products SET price = price * 0.9 WHERE category = '手机';

-- 重新开启（建议用完立即恢复）
SET SQL_SAFE_UPDATES = 1;

6.5 Column cannot be null

text
ERROR 1048 (23000): Column 'name' cannot be null

原因：向 NOT NULL 的列插入了 NULL 值

sql
-- 错误
INSERT INTO users (name, email) VALUES (NULL, 'test@test.com');   -- name 是 NOT NULL

-- 解决：给 name 赋值
INSERT INTO users (name, email) VALUES ('默认名称', 'test@test.com');

七、面试常问：DELETE vs TRUNCATE vs DROP

这是 MySQL 面试中最高频的考题之一，务必牢记！

完整对比

对比维度 DELETE TRUNCATE DROP
属于 DML DDL DDL
作用对象 数据行 数据行（全部） 整张表（含结构）
能加 WHERE ✅ 能 ❌ 不能 ❌ 不能
事务/回滚 ✅ 支持 ❌ 不支持 ❌ 不支持
触发器 ✅ 触发 ❌ 不触发 ❌ 不触发
自增重置 ❌ 不重置 ✅ 重置为1 表没了
速度 慢 快 最快
表结构 保留 保留 删除
回收空间 否（需 OPTIMIZE） 是 是

一句话总结

  • DELETE：精准删除，可回滚，可加 WHERE 条件，适合删除部分数据
  • TRUNCATE：清空表数据，不可回滚，自增重置，适合清空整张表（保留结构）
  • DROP：彻底删除表（结构+数据），不可恢复，适合删除整张表

记忆口诀

text
DELETE：外科手术刀（精准切除，可撤销）
TRUNCATE：格式化硬盘（清空数据，保留表）
DROP：炸掉整栋楼（表都没了）

八、DML 操作安全规范

sql
-- ✅ 好习惯1：UPDATE/DELETE 前先 SELECT 确认影响范围
SELECT * FROM users WHERE age < 18;   -- 先看看有多少行会被影响
DELETE FROM users WHERE age < 18;      -- 确认无误后再删除

-- ✅ 好习惯2：使用事务包裹重要操作，出错可回滚
START TRANSACTION;                     -- 开启事务

UPDATE products SET price = price * 0.9 WHERE category = '手机';
DELETE FROM orders WHERE status = 'cancelled';

-- 确认结果正确
SELECT * FROM products WHERE category = '手机';

COMMIT;    -- 确认无误，提交事务
-- ROLLBACK;  -- 如果有问题，回滚，恢复数据

-- ✅ 好习惯3：LIMIT 限制批量删除的数量
DELETE FROM logs WHERE created_at < '2024-01-01' LIMIT 1000;  -- 分批删除

-- ✅ 好习惯4：UPDATE 时用精确的主键条件
UPDATE users SET age = 26 WHERE id = 1;         -- 精准（用主键）
-- 而不是 WHERE name = '张三'（name 可能重复）

九、本章小结

操作 关键语法 注意事项
单行插入 INSERT INTO 表 (列) VALUES (值) 推荐指定列名
多行插入 VALUES (v1),(v2),(v3) 比多条SQL效率高
子查询插入 INSERT INTO ... SELECT ... 列数和类型要匹配
忽略冲突 INSERT IGNORE 冲突行被跳过
存在更新 ON DUPLICATE KEY UPDATE 触发唯一键/主键冲突时
单表更新 UPDATE 表 SET 列=值 WHERE 条件 一定要加WHERE！
多表更新 UPDATE t1 JOIN t2 ON ... SET ... 需要有关联关系
单表删除 DELETE FROM 表 WHERE 条件 一定要加WHERE！
多表删除 DELETE t1,t2 FROM t1 JOIN t2 ... 指定要删除哪张表
清空表 TRUNCATE TABLE 表名 不可回滚，自增重置

最重要的一条原则：UPDATE 和 DELETE 操作，永远先写 WHERE 条件，永远先用 SELECT 确认影响范围！
