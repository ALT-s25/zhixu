MySQL DQL 单表查询完整详解

按章节学习你的知识，慢慢成林

DQL = Data Query Language，即数据查询语言 专门用于查询表中的数据，核心关键字是 SELECT DQL 是开发中使用最频繁的操作，掌握好 DQL 是 MySQL 的核心竞争力

一、知识框架总览

text
DQL 单表查询
├── SELECT 完整语法结构
│   └── SELECT...FROM...WHERE...GROUP BY...HAVING...ORDER BY...LIMIT
│
├── 条件查询（WHERE）
│   ├── 比较运算符：= != > < >= <=
│   ├── 逻辑运算符：AND OR NOT
│   ├── 范围查询：BETWEEN...AND

│   └── 空值判断：IS NULL / IS NOT NULL
│
├── 聚合函数
│   └── COUNT / SUM / AVG / MAX / MIN
│
├── 分组查询
│   ├── GROUP BY（分组）
│   ├── HAVING（分组后过滤）
│   └── WHERE vs HAVING 区别
│
├── 排序（ORDER BY）
│   ├── ASC 升序（默认）
│   └── DESC 降序
│
├── 分页（LIMIT）

    └── FROM→WHERE→GROUP BY→HAVING→SELECT→ORDER BY→LIMIT

三、SELECT 完整语法结构

sql
SELECT   [DISTINCT] 列名1, 列名2, ...   -- ① 选择哪些列
FROM     表名                           -- ② 从哪张表
WHERE    条件                           -- ③ 行过滤条件
GROUP BY 列名                           -- ④ 按哪列分组
HAVING   分组后条件                      -- ⑤ 分组后过滤
ORDER BY 列名 [ASC|DESC]                -- ⑥ 排序
LIMIT    偏移量, 数量;                   -- ⑦ 限制返回行数

3.1 基本 SELECT

sql
-- 查询所有列（* 代表所有列，实际生产中不推荐用 *，会影响性能）
SELECT * FROM employees;

-- 查询指定列
SELECT name, department, salary   -- 只查这三列
FROM employees;

-- 给列起别名（AS 可省略）
SELECT
    name       AS 姓名,     -- 显示时列名变成"姓名"
    department AS 部门,     -- 显示时列名变成"部门"
    salary     AS 月薪      -- 显示时列名变成"月薪"
FROM employees;

-- 计算列（查询时动态计算，不修改表数据）
SELECT
    name,
    salary,
    salary * 12 AS 年薪      -- 计算年薪，结果显示为"年薪"列
FROM employees;

-- DISTINCT 去重（去除重复值）
SELECT DISTINCT department    -- 查询所有不重复的部门名称
FROM employees;
-- 结果：技术部、销售部、HR部、财务部（4行，不重复）

四、条件查询（WHERE 子句）

4.1 比较运算符

sql
-- = 等于
SELECT * FROM employees WHERE department = '技术部';    -- 查技术部员工

-- != 或 <> 不等于
SELECT * FROM employees WHERE department != '技术部';   -- 查非技术部员工
SELECT * FROM employees WHERE department <> '技术部';   -- 同上，两种写法都行

-- > 大于
SELECT * FROM employees WHERE salary > 15000;    -- 查工资大于15000的

-- < 小于
SELECT * FROM employees WHERE salary < 10000;    -- 查工资小于10000的

-- >= 大于等于
SELECT * FROM employees WHERE salary >= 15000;   -- 工资大于等于15000

-- <= 小于等于
SELECT * FROM employees WHERE age <= 28;         -- 年龄小于等于28

4.2 逻辑运算符

sql
-- AND（同时满足两个条件）
SELECT * FROM employees
WHERE department = '技术部'    -- 条件1：技术部
AND salary > 15000;            -- 条件2：工资大于15000
-- 技术部且工资>15000的员工

-- OR（满足任意一个条件）
SELECT * FROM employees
WHERE department = '技术部'   -- 条件1：技术部
OR department = '销售部';     -- 或者：销售部
-- 技术部或销售部的员工

-- NOT（取反）
SELECT * FROM employees
WHERE NOT department = '技术部';    -- 不是技术部的员工
-- 等价于：WHERE department != '技术部'

-- 组合使用（注意括号！）
SELECT * FROM employees
WHERE (department = '技术部' OR department = '销售部')    -- 技术部或销售部
AND salary > 10000;                                        -- 且工资大于10000
-- 括号很重要，没有括号逻辑会不同

4.3 BETWEEN...AND 范围查询

sql
-- BETWEEN 下限 AND 上限（包含两端，即闭区间 [下限, 上限]）
SELECT * FROM employees
WHERE salary BETWEEN 10000 AND 20000;    -- 工资在 10000~20000 之间（含两端）
-- 等价于：WHERE salary >= 10000 AND salary <= 20000

-- 日期也可以用 BETWEEN
SELECT * FROM employees
WHERE hire_date BETWEEN '2020-01-01' AND '2021-12-31';    -- 2020~2021年入职的

-- NOT BETWEEN（不在范围内）
SELECT * FROM employees
WHERE salary NOT BETWEEN 10000 AND 20000;    -- 工资不在 10000~20000 之间

4.4 IN / NOT IN 集合查询

sql
-- IN（在集合中）
SELECT * FROM employees
WHERE department IN ('技术部', '销售部', 'HR部');    -- 在这三个部门之一
-- 等价于：WHERE department='技术部' OR department='销售部' OR department='HR部'

-- NOT IN（不在集合中）
SELECT * FROM employees
WHERE department NOT IN ('技术部', '财务部');    -- 不在技术部和财务部

-- IN 配合数字
SELECT * FROM employees
WHERE id IN (1, 3, 5, 7);    -- 查询 id 为 1、3、5、7 的员工

4.5 LIKE 模糊查询

% 匹配任意多个字符（包括0个） _ 匹配任意单个字符

sql
-- % 匹配任意多个字符

-- 以"张"开头（姓张的）
SELECT * FROM employees WHERE name LIKE '张%';

-- 以"部"结尾
SELECT * FROM employees WHERE department LIKE '%部';    -- 所有带"部"字的部门

-- 包含"技"字（任意位置）
SELECT * FROM employees WHERE department LIKE '%技%';

-- _ 匹配单个字符

-- 名字是2个字（一个下划线匹配一个字符）
SELECT * FROM employees WHERE name LIKE '__';    -- 两个下划线 = 名字2个字
-- 注：中文一个字算一个字符（UTF8mb4下）

-- 以"技"开头，后面跟两个字符
SELECT * FROM employees WHERE department LIKE '技__';    -- 匹配"技术部"等3字部门名

-- 组合示例：邮箱包含 @gmail
-- SELECT * FROM users WHERE email LIKE '%@gmail%';

-- ⚠️ LIKE 注意：
-- 1. 通配符在前面（如 '%张'）会导致索引失效，查询变慢
-- 2. 推荐使用全文索引（FULLTEXT）来替代 LIKE '%关键词%'

4.6 IS NULL / IS NOT NULL

NULL 表示未知值，不能用 = NULL，必须用 IS NULL

sql
-- 查询工资为 NULL 的员工（刘二的工资是 NULL）
SELECT * FROM employees
WHERE salary IS NULL;    -- 工资为 NULL（注意：不能写 salary = NULL，查不到！）

-- 查询工资不为 NULL 的员工
SELECT * FROM employees
WHERE salary IS NOT NULL;    -- 工资有值的员工

-- ⚠️ 常见错误
SELECT * FROM employees WHERE salary = NULL;    -- 错误写法！永远查不到结果
SELECT * FROM employees WHERE salary != NULL;   -- 错误写法！永远查不到结果
-- NULL 不能用 = 或 != 比较，只能用 IS NULL / IS NOT NULL

五、聚合函数

聚合函数对一组行进行计算，返回单个值 注意：聚合函数自动忽略 NULL 值（COUNT(*) 除外）

5.1 COUNT 计数

sql
-- COUNT(*) 统计总行数（包括 NULL 行）
SELECT COUNT(*) AS 总员工数 FROM employees;    -- 结果：12（所有行）

-- COUNT(列名) 统计该列非 NULL 的行数
SELECT COUNT(salary) AS 有工资记录的人数 FROM employees;    -- 结果：11（刘二工资是NULL，不算）

-- COUNT(DISTINCT 列名) 统计不重复的值的个数
SELECT COUNT(DISTINCT department) AS 部门数量 FROM employees;    -- 结果：4（技术部/销售部/HR部/财务部）

-- 配合 WHERE 统计满足条件的行数
SELECT COUNT(*) AS 技术部人数
FROM employees
WHERE department = '技术部';    -- 只统计技术部

5.2 SUM 求和

sql
-- SUM 计算总和（自动忽略 NULL）
SELECT SUM(salary) AS 工资总额
FROM employees;    -- 所有非NULL工资之和

-- 配合 WHERE 计算技术部工资总和
SELECT SUM(salary) AS 技术部工资总额
FROM employees
WHERE department = '技术部';

-- 计算年薪总额（动态计算）
SELECT SUM(salary * 12) AS 全公司年薪总额
FROM employees;

5.3 AVG 平均值

sql
-- AVG 计算平均值（自动忽略 NULL，所以只用11个人的工资算平均）
SELECT AVG(salary) AS 平均工资
FROM employees;

-- 配合条件
SELECT AVG(salary) AS 技术部平均工资
FROM employees
WHERE department = '技术部';

-- 结合 ROUND 保留2位小数
SELECT ROUND(AVG(salary), 2) AS 平均工资（保留2位小数）
FROM employees;

5.4 MAX 和 MIN

sql
-- MAX 最大值
SELECT MAX(salary) AS 最高工资 FROM employees;    -- 22000.00

-- MIN 最小值（NULL 被忽略，不算最小值）
SELECT MIN(salary) AS 最低工资 FROM employees;    -- 7500.00（NULL被忽略了）

-- 同时查多个聚合函数
SELECT
    COUNT(*)         AS 总人数,
    COUNT(salary)    AS 有薪资人数,
    SUM(salary)      AS 工资总额,
    ROUND(AVG(salary), 2) AS 平均工资,
    MAX(salary)      AS 最高工资,
    MIN(salary)      AS 最低工资
FROM employees;

六、分组查询（GROUP BY + HAVING）

6.1 GROUP BY 分组

GROUP BY 把相同值的行归为一组，再对每组进行聚合计算

sql
-- 语法
SELECT 分组列, 聚合函数
FROM 表名
GROUP BY 分组列;
sql
-- 示例1：按部门统计人数
SELECT
    department AS 部门,          -- 分组依据列
    COUNT(*) AS 人数              -- 对每个分组统计行数
FROM employees
GROUP BY department;             -- 按部门分组
-- 结果：每个部门一行，显示部门名和人数

-- 示例2：按部门统计工资情况
SELECT
    department      AS 部门,
    COUNT(*)        AS 人数,
    SUM(salary)     AS 工资总额,
    ROUND(AVG(salary), 2) AS 平均工资,
    MAX(salary)     AS 最高工资,
    MIN(salary)     AS 最低工资
FROM employees
GROUP BY department;    -- 按部门分组，对每组分别计算

-- 示例3：按性别统计平均年龄
SELECT
    gender      AS 性别,
    COUNT(*)    AS 人数,
    AVG(age)    AS 平均年龄
FROM employees
GROUP BY gender;    -- 按性别分组

-- 示例4：按多列分组（按部门+城市统计）
SELECT
    department AS 部门,
    city       AS 城市,
    COUNT(*)   AS 人数
FROM employees
GROUP BY department, city;    -- 按部门和城市两列同时分组
-- 结果：每个（部门+城市）组合一行

-- ⚠️ 重要规则：
-- SELECT 中出现的列，要么在 GROUP BY 中，要么是聚合函数！
-- 错误示例：
-- SELECT name, department, COUNT(*) FROM employees GROUP BY department;
-- name 没有在 GROUP BY 中，也不是聚合函数，MySQL 会报错（严格模式下）

6.2 HAVING：分组后过滤

HAVING 是对 GROUP BY 分组后的结果进行过滤（相当于对分组结果的 WHERE）

sql
-- 语法
SELECT 分组列, 聚合函数
FROM 表名
GROUP BY 分组列
HAVING 聚合函数条件;
sql
-- 示例1：查询人数大于2的部门
SELECT
    department AS 部门,
    COUNT(*) AS 人数
FROM employees
GROUP BY department
HAVING COUNT(*) > 2;    -- 分组后，只保留人数>2的部门

-- 示例2：查询平均工资超过12000的部门
SELECT
    department AS 部门,
    ROUND(AVG(salary), 2) AS 平均工资
FROM employees
GROUP BY department
HAVING AVG(salary) > 12000;    -- 只显示平均工资>12000的部门

-- 示例3：WHERE + GROUP BY + HAVING 同时使用
-- 查询：技术部和销售部中，人数超过2人的部门的平均工资
SELECT
    department AS 部门,
    COUNT(*) AS 人数,
    ROUND(AVG(salary), 2) AS 平均工资
FROM employees
WHERE department IN ('技术部', '销售部')    -- ① 先过滤：只看技术部和销售部
GROUP BY department                         -- ② 按部门分组
HAVING COUNT(*) > 2;                        -- ③ 分组后，只保留人数>2的部门

6.3 WHERE vs HAVING 核心区别

对比维度 WHERE HAVING
执行时机 GROUP BY 之前（对原始数据过滤） GROUP BY 之后（对分组结果过滤）
能用聚合函数吗 ❌ 不能（此时还没分组） ✅ 能（已经分组，有聚合结果）
作用对象 原始表的每一行 分组后的每一组
性能 先过滤再分组（效率高） 先分组再过滤（效率低一点）
sql
-- 错误示例：WHERE 中不能用聚合函数
-- SELECT department, COUNT(*) FROM employees
-- WHERE COUNT(*) > 2    ← 错误！WHERE 在分组前执行，没有聚合结果
-- GROUP BY department;

-- 正确写法：用 HAVING
SELECT department, COUNT(*) AS 人数
FROM employees
GROUP BY department
HAVING COUNT(*) > 2;    -- 正确！HAVING 在分组后执行

-- 能用 WHERE 的就用 WHERE（效率更高）
SELECT department, COUNT(*) AS 人数
FROM employees
WHERE salary > 10000       -- 先用 WHERE 过滤掉工资≤10000的（原始行）
GROUP BY department
HAVING COUNT(*) > 1;       -- 再用 HAVING 过滤分组结果

七、排序（ORDER BY）

7.1 单列排序

sql
-- 语法
SELECT 列 FROM 表名 ORDER BY 列名 [ASC|DESC];
-- ASC：升序（从小到大，默认值，可以不写）
-- DESC：降序（从大到小）
sql
-- 升序（默认，ASC可以省略）
SELECT name, salary FROM employees
ORDER BY salary ASC;      -- 工资从低到高排序（ASC 可以省略）

-- 降序
SELECT name, salary FROM employees
ORDER BY salary DESC;     -- 工资从高到低排序

-- 按日期排序（最新的在前）
SELECT name, hire_date FROM employees
ORDER BY hire_date DESC;  -- 入职日期最新的排前面

-- 按字符串排序（汉字按拼音/编码排序）
SELECT name, department FROM employees
ORDER BY name ASC;    -- 按姓名拼音升序

7.2 多列排序

先按第一列排序，第一列相同时，再按第二列排序

sql
-- 先按部门升序，同部门内按工资降序
SELECT name, department, salary
FROM employees
ORDER BY
    department ASC,     -- 第一排序字段：部门升序
    salary DESC;        -- 第二排序字段：工资降序（部门相同时）

-- 先按城市排序，同城市按年龄升序
SELECT name, city, age
FROM employees
ORDER BY
    city ASC,           -- 先按城市排
    age ASC;            -- 城市相同时按年龄排

-- 按表达式排序（按年薪从高到低）
SELECT name, salary, salary * 12 AS 年薪
FROM employees
ORDER BY salary * 12 DESC;    -- 按年薪降序，可以直接用表达式
-- 也可以用别名：ORDER BY 年薪 DESC

-- ⚠️ NULL 值排序：
-- MySQL 中 NULL 在 ASC 排序时排在最前面
-- NULL 在 DESC 排序时排在最后面
SELECT name, salary FROM employees ORDER BY salary ASC;   -- NULL 排最前
SELECT name, salary FROM employees ORDER BY salary DESC;  -- NULL 排最后

八、分页查询（LIMIT）

一次性返回所有数据太多，用 LIMIT 控制每次返回的行数 常用于前端分页展示

8.1 LIMIT 语法

sql
-- 方式1：LIMIT count（只限制数量，从第1行开始）
SELECT * FROM employees LIMIT 5;    -- 只返回前5行

-- 方式2：LIMIT offset, count（跳过 offset 行，取 count 行）
SELECT * FROM employees LIMIT 0, 5;    -- 跳过0行，取5行（第1页）
SELECT * FROM employees LIMIT 5, 5;    -- 跳过5行，取5行（第2页）
SELECT * FROM employees LIMIT 10, 5;   -- 跳过10行，取5行（第3页）

-- 方式3：LIMIT count OFFSET offset（MySQL 5.7+ 支持，更清晰）
SELECT * FROM employees LIMIT 5 OFFSET 0;    -- 第1页
SELECT * FROM employees LIMIT 5 OFFSET 5;    -- 第2页

8.2 分页公式

text
第 N 页的查询（每页显示 pageSize 条）：
LIMIT (N-1)*pageSize, pageSize

例：每页5条，查第3页：
LIMIT (3-1)*5, 5  →  LIMIT 10, 5

8.3 完整分页示例

sql
-- 场景：员工列表分页，按工资降序，每页5条

-- 第1页（第1~5条）
SELECT name, department, salary
FROM employees
ORDER BY salary DESC    -- 按工资降序
LIMIT 0, 5;             -- 偏移0，取5条

-- 第2页（第6~10条）
SELECT name, department, salary
FROM employees
ORDER BY salary DESC
LIMIT 5, 5;             -- 偏移5，取5条

-- 第3页（第11~12条，不足5条也没关系）
SELECT name, department, salary
FROM employees
ORDER BY salary DESC
LIMIT 10, 5;            -- 偏移10，取5条（实际只有2条）

-- 结合 COUNT 实现前端分页
-- 第1步：查总条数（前端用来计算总页数）
SELECT COUNT(*) AS 总条数 FROM employees;    -- 结果：12

-- 第2步：查当前页的数据
SELECT name, department, salary
FROM employees
ORDER BY salary DESC
LIMIT 0, 5;    -- 第1页数据
-- 前端：总页数 = CEIL(12/5) = 3 页

九、SQL 执行顺序

理解执行顺序，才能正确写出复杂查询，也是面试高频考点

9.1 执行顺序图

text
书写顺序：SELECT → FROM → WHERE → GROUP BY → HAVING → ORDER BY → LIMIT
执行顺序：FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT

① FROM      ：确定数据来源（从哪张表）
② WHERE     ：过滤原始行（聚合前）
③ GROUP BY  ：按列分组
④ HAVING    ：过滤分组结果（聚合后）
⑤ SELECT    ：选择输出的列（计算别名）
⑥ ORDER BY  ：排序（此时可以用 SELECT 中定义的别名）
⑦ LIMIT     ：截取指定行数

9.2 执行顺序示例解析

sql
-- 完整示例：统计每个部门工资>10000员工中，平均工资>13000的部门
SELECT
    department AS 部门,          -- ⑤ SELECT：选择列，定义别名
    COUNT(*) AS 人数,
    ROUND(AVG(salary), 2) AS 平均工资
FROM employees                   -- ① FROM：先扫描 employees 表
WHERE salary > 10000             -- ② WHERE：过滤工资>10000的行
GROUP BY department              -- ③ GROUP BY：按部门分组
HAVING AVG(salary) > 13000       -- ④ HAVING：过滤平均工资>13000的分组
ORDER BY 平均工资 DESC            -- ⑥ ORDER BY：按平均工资降序（可以用别名）
LIMIT 3;                         -- ⑦ LIMIT：只取前3条

9.3 执行顺序的实际影响

sql
-- 问题1：ORDER BY 能用 SELECT 中定义的别名吗？
-- 答：能！因为 ORDER BY 在 SELECT 之后执行

SELECT name, salary * 12 AS 年薪
FROM employees
ORDER BY 年薪 DESC;    -- ✅ 可以用别名"年薪"

-- 问题2：WHERE 能用 SELECT 中定义的别名吗？
-- 答：不能！因为 WHERE 在 SELECT 之前执行

-- 错误写法：
-- SELECT name, salary * 12 AS 年薪
-- FROM employees
-- WHERE 年薪 > 100000;    -- ❌ 报错！WHERE 时别名还没定义

-- 正确写法：用原始表达式
SELECT name, salary * 12 AS 年薪
FROM employees
WHERE salary * 12 > 100000;    -- ✅ 用原始表达式，不用别名

-- 问题3：HAVING 能用 SELECT 中定义的别名吗？
-- 答：MySQL 中可以，但严格来说不规范（部分数据库不支持）
SELECT department, AVG(salary) AS 平均工资
FROM employees
GROUP BY department
HAVING 平均工资 > 10000;    -- MySQL 中可以，但建议写成 HAVING AVG(salary) > 10000

十、综合实战示例

sql
-- 实战1：查询技术部中工资前3名的员工信息
SELECT name, department, salary
FROM employees
WHERE department = '技术部'    -- 只看技术部
ORDER BY salary DESC            -- 工资从高到低
LIMIT 3;                        -- 只取前3名

-- 实战2：统计各部门男女员工人数
SELECT
    department AS 部门,
    gender     AS 性别,
    COUNT(*)   AS 人数
FROM employees
GROUP BY department, gender    -- 按部门+性别分组
ORDER BY department, gender;   -- 排序

-- 实战3：查询工资高于本公司平均工资的员工（子查询，DQL进阶用法）
SELECT name, salary
FROM employees
WHERE salary > (
    SELECT AVG(salary)    -- 子查询：计算平均工资
    FROM employees
)
ORDER BY salary DESC;

-- 实战4：各城市平均工资排行（过滤掉人数不足2人的城市）
SELECT
    city AS 城市,
    COUNT(*) AS 人数,
    ROUND(AVG(salary), 2) AS 平均工资
FROM employees
WHERE salary IS NOT NULL         -- 排除工资为NULL的行
GROUP BY city                    -- 按城市分组
HAVING COUNT(*) >= 2             -- 只保留人数>=2的城市
ORDER BY 平均工资 DESC;           -- 按平均工资降序排列

-- 实战5：分页查询，第2页，每页4条，按入职时间降序
SELECT
    id,
    name,
    department,
    hire_date
FROM employees
ORDER BY hire_date DESC    -- 最新入职的排前面
LIMIT 4, 4;                -- 第2页：跳过4行，取4行

十一、常见报错及解决方法

11.1 Expression not in GROUP BY

text
ERROR 1055 (42000): Expression #1 of SELECT list is not in GROUP BY clause
and contains nonaggregated column...

原因：SELECT 中的列没有在 GROUP BY 中，也没有用聚合函数

sql
-- 错误
SELECT name, department, COUNT(*)
FROM employees
GROUP BY department;    -- name 没有在 GROUP BY 中，报错（严格模式）

-- 解决方法1：把 name 加到 GROUP BY
SELECT name, department, COUNT(*)
FROM employees
GROUP BY department, name;    -- name 也加入分组（但结果会变）

-- 解决方法2：用聚合函数（如果想看任意一个name）
SELECT MAX(name) AS 代表姓名, department, COUNT(*)
FROM employees
GROUP BY department;

-- 解决方法3：关闭严格模式（不推荐，会返回不确定的值）
-- SET sql_mode = '';    -- 临时方案，不建议

11.2 Unknown column in HAVING

text
ERROR 1054 (42S22): Unknown column '平均工资' in 'having clause'

原因：某些 MySQL 版本 HAVING 中不能用 SELECT 的别名

sql
-- 可能报错的写法
SELECT department, AVG(salary) AS 平均工资
FROM employees
GROUP BY department
HAVING 平均工资 > 10000;    -- 某些版本报错

-- 安全写法：直接用聚合表达式
SELECT department, AVG(salary) AS 平均工资
FROM employees
GROUP BY department
HAVING AVG(salary) > 10000;    -- 直接写 AVG(salary)，不用别名

11.3 LIKE 查不到中文

原因：字符集设置问题，或通配符使用不当

sql
-- 检查字符集
SHOW VARIABLES LIKE 'character_set%';    -- 查看字符集设置

-- 确保表和字段是 utf8mb4
ALTER TABLE employees CONVERT TO CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- LIKE 中文查询（确保字符集正确后正常使用）
SELECT * FROM employees WHERE name LIKE '张%';    -- 查姓张的

11.4 ORDER BY 结果不稳定（翻页数据重复）

原因：排序字段有重复值时，MySQL 的排序结果是不确定的

sql
-- 不稳定的排序（salary 可能有相同值，每次查询顺序不同）
SELECT * FROM employees ORDER BY salary DESC LIMIT 5;

-- 加上唯一键作为二级排序，保证稳定
SELECT * FROM employees ORDER BY salary DESC, id ASC LIMIT 5;
-- 工资相同时，按 id 升序排，保证每次结果一致

十二、面试常问

12.1 WHERE vs HAVING 区别（极高频）

见第六节第三小节，记住这个核心结论：

  • WHERE：分组前过滤，不能用聚合函数
  • HAVING：分组后过滤，可以用聚合函数

12.2 COUNT(*) vs COUNT(列名) 区别

text
COUNT(*) ：统计所有行数，包括 NULL 行
COUNT(列名)：统计该列非 NULL 的行数，自动忽略 NULL
COUNT(1) ：等价于 COUNT(*)，统计所有行（1是常量，不会是NULL）
sql
SELECT
    COUNT(*) AS 全部人数,           -- 12（包含工资为NULL的刘二）
    COUNT(salary) AS 有薪资人数,    -- 11（刘二被排除）
    COUNT(1) AS 也是全部人数        -- 12（COUNT(1) = COUNT(*)）
FROM employees;

12.3 SQL 执行顺序（高频）

text
FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT
记忆口诀：
"大家（FROM）围着（WHERE）桌子（GROUP BY）喝（HAVING）酒
         选（SELECT）位置（ORDER BY）坐（LIMIT）下来"

12.4 DISTINCT 注意事项

sql
-- DISTINCT 作用于所有列（不是只作用于紧跟的那一列）
SELECT DISTINCT department, gender FROM employees;
-- 去重的是（department + gender）的组合，不是单独去重 department

-- 聚合函数中的 DISTINCT
SELECT COUNT(DISTINCT department) FROM employees;    -- 统计不重复的部门数
SELECT SUM(DISTINCT salary) FROM employees;          -- 只加不重复的工资值（奇怪用法，了解即可）

12.5 LIMIT 大偏移量性能问题

sql
-- 问题：LIMIT 100000, 10 很慢（要先扫描10万行再丢弃）
SELECT * FROM employees ORDER BY id LIMIT 100000, 10;    -- 慢！

-- 优化方案：用 WHERE 替代大偏移量
SELECT * FROM employees WHERE id > 100000 ORDER BY id LIMIT 10;    -- 快！
-- 前提：id 有索引，且用游标方式翻页

十三、本章小结

语法 作用 关键点
SELECT 列 选择输出列 * 低效，推荐指定列名
DISTINCT 去重 作用于所有列的组合
WHERE 条件 过滤行 聚合前执行，不能用聚合函数
BETWEEN 范围查询 闭区间，含两端
IN 集合查询 可替代多个 OR
LIKE 模糊查询 %多个字符，_单个字符
IS NULL 空值判断 不能用 = NULL
COUNT/SUM/AVG/MAX/MIN 聚合函数 自动忽略 NULL（COUNT* 除外）
GROUP BY 分组 SELECT 列要在 GROUP BY 中或是聚合函数
HAVING 分组后过滤 聚合后执行，可以用聚合函数
ORDER BY 排序 ASC升序（默认），DESC降序
LIMIT offset, count 分页 偏移量从0开始

SQL 执行顺序（最重要）：FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT 理解了执行顺序，所有别名、聚合函数使用时机的问题都迎刃而解！
